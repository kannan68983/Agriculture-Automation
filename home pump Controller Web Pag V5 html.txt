                    if(tEl) tEl.innerText = fertLabels[i];
                }

            db.ref('system/emergencyStop').on('value', (snap) => {
                if (snap.val() === true) {
                    db.ref('system/lastTripReason').once('value').then((res) => {
                        let code = res.val() || 0;
                        let desc = "Manual Emergency Stop Activated.";
                        if(code === 1) desc = "FAULT [VOLTAGE]: Grid power is out of safe bounds. Motor halted to prevent coil damage.";
                        if(code === 2) desc = "FAULT [DRY RUN]: Current dropped below minimum limit. Max auto-retries reached. Check the water source.";
                        if(code === 3) desc = "FAULT [OVERLOAD]: Current exceeded maximum limit. Max auto-retries reached. The motor may be jammed.";
                        if(code === 4) desc = "FAULT [SENSOR]: PZEM Power Monitor disconnected or unreadable.";
                        
                        document.getElementById('criticalAlertDesc').innerText = desc;
                        document.getElementById('criticalAlertOverlay').style.display = 'block';
                        
                        clearTimeout(alertTimeout);
                        alertTimeout = setTimeout(() => { clearEmergencyStop(); }, 10000);
                    });
                } else {
                    document.getElementById('criticalAlertOverlay').style.display = 'none';
                }
            });

            db.ref('system/interlockReason').on('value', (snap) => {
                let code = snap.val() || 0;
                currentInterlockReason = code;
                
                let motorProtEnabled = (currentSysState.motorConfig && currentSysState.motorConfig.enabled === true);
                if (code > 0 && !currentSysState.emergencyStop && motorProtEnabled) {
                    let desc = "Hardware Interlock Active.";
                    if(code === 1) desc = "VOLTAGE OUT OF BOUNDS: Grid power is unstable.";
                    if(code === 2) desc = "DRY RUN DETECTED: Current dropped below minimum limit.";
                    if(code === 3) desc = "OVERLOAD DETECTED: Current exceeded maximum limit.";
                    if(code === 4) desc = "SENSOR FAULT: PZEM Power Monitor disconnected.";
                    
                    document.getElementById('autoRecoveryDesc').innerText = desc;
                    document.getElementById('autoRecoveryOverlay').style.display = 'block';
                    
                    clearTimeout(interlockTimeout);
                    interlockTimeout = setTimeout(() => {
                        document.getElementById('autoRecoveryOverlay').style.display = 'none';
                    }, 10000);
                } else {
                    document.getElementById('autoRecoveryOverlay').style.display = 'none';
                }
            });

            db.ref('system/motorConfig').on('value', (snap) => {
                let conf = snap.val() || {};
                document.getElementById('prot-toggle').checked = conf.enabled || false;
                if(conf.minV) document.getElementById('prot-min-v').value = conf.minV;
                if(conf.maxV) document.getElementById('prot-max-v').value = conf.maxV;
                if(conf.minA) document.getElementById('prot-min-a').value = conf.minA;
                if(conf.maxA) document.getElementById('prot-max-a').value = conf.maxA;
            });

            db.ref('system').on('value', (snap) => {
                currentSysState = snap.val() || {};
                
                let resState = currentSysState.resume || {};
                let isResEnabled = (resState.enabled === true);
                
                document.getElementById('dash-resume-toggle').checked = isResEnabled;

                let pendingZone = currentSysState.pendingZone || 0;
                let pendingSeconds = currentSysState.pendingRemain || 0;
                
                let zName = (pendingZone > 0 && pendingZone <= 6) ? zoneLabels[pendingZone - 1] : "No Task";
                
                let zTime = "00:00";
                if (pendingZone > 0 && pendingSeconds > 0) {
                    let minutes = Math.floor(pendingSeconds / 60);
                    let seconds = pendingSeconds % 60;
                    zTime = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
                }
                
                let zStatus = isResEnabled ? "ACTIVE" : "INACTIVE";
                let statusColor = isResEnabled ? "var(--fert-color)" : "var(--danger)";
                
                if (pendingZone === 0 || pendingSeconds === 0) {
                     zStatus = "--";
                     statusColor = "var(--text-muted)";
                }

                document.getElementById('dash-pending-zone').innerText = zName;
                document.getElementById('dash-pending-time').innerText = zTime;
                document.getElementById('dash-resume-status').innerText = zStatus;
                document.getElementById('dash-resume-status').style.color = statusColor;
            });

            db.ref('schedules/backwash').on('value', (snap) => {
                let bw = snap.val() || {};
                document.getElementById('bw-toggle').checked = bw.active || false;
                document.getElementById('bw-time').value = `${String(bw.hour||0).padStart(2,'0')}:${String(bw.minute||0).padStart(2,'0')}`;

                let data = snap.val() || {};
                localSchedules = data;
                
                const listDiv = document.getElementById('active-schedules-list');
                listDiv.innerHTML = "";

                for (let s=0; s<3; s++) {
                    let key = 'slot_' + s;
                    let sched = data[key];
                    if (!sched || (!sched.hour && !sched.starts)) continue;
                    
                    hasSchedules = true;
                    
                    let stText = "";
                    if (sched.starts && sched.starts.length > 0) {
                        let stArray = [];
                        sched.starts.forEach(st => {
                            if (st) stArray.push(`${String(st.hour).padStart(2, '0')}:${String(st.minute).padStart(2, '0')}`);
                        });
                        stText = stArray.join(" &bull; ");

                        stText = `${String(sched.hour || 6).padStart(2, '0')}:${String(sched.minute || 0).padStart(2, '0')}`;

                    for(let i=0; i<7; i++) {
                        if((sched.days & dayVals[i]) !== 0) activeDays.push(dayNames[i]);

'checked' : ''} onchange="toggleScheduleActive('${key}', this.checked)"><span class="slider"></span></label></div><div class="schedule-info"><p><strong>DAYS:</strong> ${activeDays.length === 7 ? "Everyday" : activeDays.join(", ")}</p><p><strong>STATUS:</strong> ${sched.active ?             setInterval(() => {
                let sys = currentSysState;
                if (!sys) return;
                
                let nowSec = Math.floor(Date.now() / 1000);
                let isOnline = (nowSec - (sys.heartbeat || 0)) < 25;
                
                let motorProtEnabled = (sys.motorConfig && sys.motorConfig.enabled === true);
                let isPowerCut = motorProtEnabled && (currentInterlockReason > 0 || (sys.voltage !== undefined && sys.voltage < 100));
                let systemHalted = motorProtEnabled && (isPowerCut || currentInterlockReason > 0);
                
                let activeIrr = sys.activeIrrZone || 0;
                let activeFert = sys.activeFertZone || 0;
                let manIrr = sys.manualIrrZone || 0;
                let manFert = sys.manualFertZone || 0;
                
                let p1Active = (sys.pump1Active === true) && !isPowerCut;
                let fPumpActive = (sys.fertPumpActive === true) && !isPowerCut;
                let isBwActive = sys.isBackwashActive === true;
                
                let irrStop = sys.activeIrrStop || 0;
                let fertStop = sys.activeFertStop || 0;
                
                let calcIrrRem = irrStop > 0 ? irrStop - nowSec : 0;
                if (calcIrrRem < 0) calcIrrRem = 0;

                let calcFertRem = fertStop > 0 ? fertStop - nowSec : 0;
                if (calcFertRem < 0) calcFertRem = 0;

                if (systemHalted) {
                    if (!isTimerFrozen) {
                        frozenIrrRem = (sys.pendingRemain && sys.pendingRemain > 0) ? sys.pendingRemain : calcIrrRem;
                        frozenFertRem = calcFertRem; 
                        isTimerFrozen = true;
                    }
                } else {
                    isTimerFrozen = false;
                }

                let actualIrrRem = isTimerFrozen ? frozenIrrRem : calcIrrRem;
                let actualFertRem = isTimerFrozen ? frozenFertRem : calcFertRem;

                let displayIrrZone = activeIrr > 0 ? activeIrr : (manIrr > 0 ? manIrr : 0);
                let displayFertZone = activeFert > 0 ? activeFert : (manFert > 0 ? manFert : 0);

                // --- TIMER PROGRESS CALCULATION ---
                if (displayIrrZone !== lastIrrZone) {
                    lastIrrZone = displayIrrZone;
                    maxIrrSec = actualIrrRem;
                } else if (actualIrrRem > maxIrrSec) {
                    maxIrrSec = actualIrrRem;
                }

                if (displayFertZone !== lastFertZone) {
                    lastFertZone = displayFertZone;
                    maxFertSec = actualFertRem;
                } else if (actualFertRem > maxFertSec) {
                    maxFertSec = actualFertRem;
                }

                let irrProgress = 0;
                if (maxIrrSec > 0 && actualIrrRem > 0) {
                    irrProgress = ((maxIrrSec - actualIrrRem) / maxIrrSec) * 100;
                } else {
                    irrProgress = 0;
                    maxIrrSec = 0;
                }

                let fertProgress = 0;
                if (maxFertSec > 0 && actualFertRem > 0) {
                    fertProgress = ((maxFertSec - actualFertRem) / maxFertSec) * 100;
                } else {
                    fertProgress = 0;
                    maxFertSec = 0;
                }
                // ----------------------------------

                let hb = document.getElementById('heartbeat-display');
                let wifiDisplay = document.getElementById('wifi-display');
                
                let voltUI = document.getElementById('pzem-volt');
                let currUI = document.getElementById('pzem-curr');

                if(isOnline) {
                    hb.className = "connection-status status-online";
                    hb.innerHTML = `<div class="pulse-dot"></div> ONLINE`;
                    
                    let rssi = sys.rssi || 0;
                    let wifiColor = "var(--text-muted)";
                    if (rssi < 0) {
                        if (rssi > -60) { wifiColor = "var(--fert-color)"; }
                        else if (rssi > -75) { wifiColor = "var(--warning)"; }
                        else { wifiColor = "var(--danger)"; }
                        wifiDisplay.innerHTML = `<i class="fa-solid fa-wifi" style="color:${wifiColor}"></i> ${rssi} dBm`;
                    } else {
                        wifiDisplay.innerHTML = `<i class="fa-solid fa-wifi"></i> -- dBm`;
                    }

                    if(sys.voltage === undefined || sys.voltage < 0) {
                        voltUI.innerHTML = `<span class="power-error"><i class="fa-solid fa-plug-circle-xmark"></i> ERROR</span>`;
                        currUI.innerHTML = `<span style="opacity:0.3;">-- A</span>`;
                    } else {
                        voltUI.innerText = `${parseFloat(sys.voltage).toFixed(1)} V`;
                        currUI.innerText = `${parseFloat(sys.current).toFixed(2)} A`;
                        
                        if (motorProtEnabled) {
                            if (sys.voltage < sys.motorConfig.minV || sys.voltage > sys.motorConfig.maxV) voltUI.className = "power-val power-error";
                            else voltUI.className = "power-val";
                            
                            if (p1Active && sys.current >= 0) {
                                if (sys.current < sys.motorConfig.minA || sys.current > sys.motorConfig.maxA) currUI.className = "power-val power-error";
                                else currUI.className = "power-val";
                            } else currUI.className = "power-val";
                        } else {
                            voltUI.className = "power-val";
                            currUI.className = "power-val";
                        }
                    }
                } else {
                    hb.className = "connection-status status-offline";
                    hb.innerHTML = `<i class="fa-solid fa-triangle-exclamation"></i> OFFLINE`;
                    wifiDisplay.innerHTML = `<i class="fa-solid fa-wifi" style="color:var(--text-muted); opacity:0.4;"></i> <span style="opacity:0.4;">-- dBm</span>`;
                    
                    voltUI.innerHTML = `<span class="power-error"><i class="fa-solid fa-triangle-exclamation"></i> OFFLINE</span>`;
                    currUI.innerHTML = `<span style="opacity:0.3;">-- A</span>`;
                    voltUI.className = "power-val";
                    currUI.className = "power-val";
                }

                // HERO DASHBOARD - Updates to show info in top box directly
                const hero = document.getElementById('hero-dashboard');
                if (!sys.emergencyStop) {
                    if (isPowerCut) {
                        hero.className = "telemetry-card telemetry-paused";
                        document.getElementById('hero-title').innerText = "POWER FAILURE";
                        document.getElementById('hero-title').style.color = "var(--danger)";

                        document.getElementById('hero-subtitle').innerText = "Flushing filters and pipelines.";

                        hero.className = "telemetry-card telemetry-active";
                        document.getElementById('hero-title').innerText = (displayIrrZone > 0 ? `${zoneLabels[displayIrrZone-1].toUpperCase()} ` : "") + (displayFertZone > 0 ? `| ${fertLabels[displayFertZone-1].toUpperCase()}` : "");
                        document.getElementById('hero-title').style.color = "var(--text-main)";
                        document.getElementById('hero-subtitle').innerText = "Live System Telemetry";
                    } else {
                        hero.className = "telemetry-card";
                        document.getElementById('hero-title').innerText = "STANDBY";
                        document.getElementById('hero-title').style.color = "var(--text-main)";
                        document.getElementById('hero-subtitle').innerText = "Awaiting schedule execution.";

                document.getElementById('motor-main-ind').innerHTML = p1Active ? `<i class="fa-solid fa-fan"></i> Main Pump: ON` : `<i class="fa-solid fa-water"></i> Main Pump: OFF`;
                
                document.getElementById('motor-fert-ind').className = fPumpActive ? "motor-indicator motor-on m-fert" : "motor-indicator";
                document.getElementById('motor-fert-ind').innerHTML = fPumpActive ? `<i class="fa-solid fa-gear"></i> Fert Pump: ON` : `<i class="fa-solid fa-flask"></i> Fert Pump: OFF`;

                    toggle.disabled = isBwActive || isPowerCut;

                    if (isZoneActive) {
                        if (isPowerCut) {

                            card.classList.remove("pump-running");

                            card.className = "control-card active-irr";
                            document.getElementById(`status-irr-${i}`).innerText = "FLOW ACTIVE";
                            if (p1Active) card.classList.add("pump-running"); else card.classList.remove("pump-running");
                        }
                    } else {
                        card.className = "control-card";
                        document.getElementById(`status-irr-${i}`).innerText = "STANDBY";
                    }
                }

                let disableFerts = (displayIrrZone === 0 || isBwActive || isPowerCut);
                for(let i=1; i<=4; i++) {
                    let isFertActive = (displayFertZone === i);
                    let toggle = document.getElementById(`toggle-fert-${i}`);
                    toggle.checked = isFertActive;
                    toggle.disabled = disableFerts;

                    let card = document.getElementById(`card-fert-${i}`);
                    if (isFertActive) {
                        if (isPowerCut) {
                            card.className = "control-card paused";
                            document.getElementById(`status-fert-${i}`).innerText = "POWER FAILURE";
                            card.classList.remove("pump-running");
                        } else {
                            card.className = "control-card active-fert";
                            document.getElementById(`status-fert-${i}`).innerText = "INJECTING";
                            if (fPumpActive) card.classList.add("pump-running"); else card.classList.remove("pump-running");

                        card.className = "control-card";

                            card.style.opacity = "0.4";
                            card.style.pointerEvents = "none";
                            document.getElementById(`status-fert-${i}`).innerText = "DISABLED";

                            card.style.pointerEvents = "auto";

                    }

                let boxWrapper = document.getElementById('standalone-timers');
                let boxI = document.getElementById('timer-box-irr');

                let tmrF = document.getElementById('live-timer-fert');
                let nameF = document.getElementById('timer-name-fert');
                let pFert = document.getElementById('progress-fert');

                let isTimerVisible = false;

                if (displayIrrZone > 0 && !sys.emergencyStop && !isBwActive) {
                    boxI.style.display = "flex";
                    isTimerVisible = true;
                    nameI.innerText = zoneLabels[displayIrrZone - 1];
                    
                    if (pIrr) pIrr.style.width = irrProgress + "%";
                    
                    if (systemHalted) {
                        boxI.className = "separate-timer-box t-paused";

                        if (pIrr) pIrr.style.background = "var(--danger)";
                    } else if (irrStop > 0) {
                        boxI.className = "separate-timer-box t-irr";
                        if(actualIrrRem > 0) {
                            tmrI.innerText = `${Math.floor(actualIrrRem / 60).toString().padStart(2, '0')}:${(actualIrrRem % 60).toString().padStart(2, '0')}`;
                            lblI.innerText = `ZONE REMAINING`;
                            tmrI.style.color = "var(--primary)";
                            if (pIrr) pIrr.style.background = "var(--primary)";

                            lblI.innerText = `FINISHING...`;

                        lblI.innerText = `ZONE ACTIVE`;

                }

                if (displayFertZone > 0 && !sys.emergencyStop && !isBwActive) {
                    boxF.style.display = "flex";
                    isTimerVisible = true;
                    nameF.innerText = fertLabels[displayFertZone - 1];
                    
                    if (pFert) pFert.style.width = fertProgress + "%";

                    if (systemHalted) {
                        boxF.className = "separate-timer-box t-paused";
                        lblF.innerText = (currentInterlockReason > 0) ? `PAUSED (INTERLOCK)` : `PAUSED (POWER FAIL)`;
                        tmrF.innerText = `${Math.floor(actualFertRem / 60).toString().padStart(2, '0')}:${(actualFertRem % 60).toString().padStart(2, '0')}`;
                        tmrF.style.color = "var(--danger)";
                        if (pFert) pFert.style.background = "var(--danger)";
                    } else if (fertStop > 0) {
                        boxF.className = "separate-timer-box t-fert";
                        if(actualFertRem > 0) {
                            tmrF.innerText = `${Math.floor(actualFertRem / 60).toString().padStart(2, '0')}:${(actualFertRem % 60).toString().padStart(2, '0')}`;
                            lblF.innerText = `COMP REMAINING`;
                            tmrF.style.color = "var(--fert-color)";
                            if (pFert) pFert.style.background = "var(--fert-color)";
                        } else {
                            lblF.innerText = `FINISHING...`;
                            tmrF.innerText = "00:00";
                        }
                    } else {
                        boxF.className = "separate-timer-box t-fert";
                        lblF.innerText = `COMP ACTIVE`;
                        tmrF.innerText = "CONT.";
                        tmrF.style.color = "var(--fert-color)";
                        if (pFert) pFert.style.width = "100%";
                    }
                } else {
                    boxF.style.display = "none";
                }

                boxWrapper.style.display = isTimerVisible ? "flex" : "none";

            }, 1000);
        }

                pendingManualType = "fert";
                pendingManualId = compNum;

                document.getElementById('manualTimerConfirmBtn').style.background = "var(--fert-color)";
                document.getElementById('manualTimerOverlay').style.display = "block";
            } else {
                db.ref('system').update({ manualFertZone: 0, manualFertMins: 0, activeFertStop: 0 });
            }
        }

        function cancelManualTimer() { document.getElementById('manualTimerOverlay').style.display = "none"; }

        function confirmManualTimer() {
            let mins = parseInt(document.getElementById('manualTimerMins').value) || 0;
            currentSysState.activeIrrStop = 0;
            currentSysState.activeFertStop = 0;
            if (pendingManualType === "irr") db.ref('system').update({ manualIrrZone: pendingManualId, manualIrrMins: mins, activeIrrStop: 0 });
            else if (pendingManualType === "fert") db.ref('system').update({ manualFertZone: pendingManualId, manualFertMins: mins, activeFertStop: 0 });
            cancelManualTimer();
        }

        }

        function saveSchedule() {
            let startsArray = [];
            for(let i=0; i<5; i++) {
                let el = document.getElementById(`st-time-${i}`);
                if(el && el.value) {
                    let parts = el.value.split(':');
                    startsArray.push({ hour: parseInt(parts[0]), minute: parseInt(parts[1]) });
                }
            }
            if(startsArray.length === 0) { alert("At least one execution time required."); return; }
            
            let mDaysMask = 0;
            document.querySelectorAll('.m-day:checked').forEach(cb => mDaysMask += parseInt(cb.value));
            if(mDaysMask === 0) { alert("Select minimum 1 execution day."); return; }

            let sData = { active: true, starts: startsArray, days: mDaysMask, zones: [] };
            for(let z=0; z<6; z++) {
                let zoneObj = { waterMins: parseInt(document.getElementById(`z-irrDur-${z}`).value) || 0 };
                let validFerts = [];
                for(let f=0; f<fertCounts[z]; f++) {
                    let compEl = document.getElementById(`sel-comp-${z}-${f}`);
                    if(!compEl) continue;
                    let timeVal = document.getElementById(`sel-time-${z}-${f}`).value;
                    let startH = 0, startM = 0;
                    if (timeVal) { startH = parseInt(timeVal.split(':')[0]); startM = parseInt(timeVal.split(':')[1]); } else continue;
                    
                    let fDaysMask = 0;
                    document.querySelectorAll(`.f-day-${z}-${f}:checked`).forEach(cb => fDaysMask += parseInt(cb.value));
                    
                    validFerts.push({ comp: parseInt(compEl.value), startHour: startH, startMinute: startM, mins: parseInt(document.getElementById(`sel-mins-${z}-${f}`).value) || 0, days: fDaysMask });
                }
                if(validFerts.length > 0) zoneObj.ferts = validFerts;
                sData.zones.push(zoneObj);
            }

            let targetSlotKey = editingSlotKey;
            if (!targetSlotKey) {
                let schedulesObj = localSchedules || {};
                for(let i=0; i<3; i++) {
                    if (!schedulesObj['slot_' + i]) { targetSlotKey = 'slot_' + i; break; }
                }
                if (targetSlotKey === null) { alert("Max slots reached. Purge an old automation first."); return; }
            }

            db.ref(`schedules/${targetSlotKey}`).set(sData).then(() => { closeScheduleCreator(); });
        }
